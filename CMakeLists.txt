cmake_minimum_required(VERSION 3.20)
project(KozDbg LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(X64DBG_SDK "" CACHE PATH "x64dbg pluginsdk directory")
if(NOT EXISTS "${X64DBG_SDK}/_plugins.h")
 message(FATAL_ERROR "Set X64DBG_SDK to the supplied pluginsdk directory")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "The x32 branch requires -A Win32. Use the main branch to build either architecture.")
endif()
set(KOZ_SOURCE x86_plugin.cpp)
set(KOZ_ARCH x32)
set(KOZ_SUFFIX .dp32)
add_library(KozDbg SHARED ${KOZ_SOURCE} dialog.rc)
target_include_directories(KozDbg PRIVATE "${X64DBG_SDK}")
target_compile_definitions(KozDbg PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(KozDbg PRIVATE /W4 /WX /EHsc /analyze)
target_link_libraries(KozDbg PRIVATE "${X64DBG_SDK}/${KOZ_ARCH}dbg.lib" "${X64DBG_SDK}/${KOZ_ARCH}bridge.lib" comctl32 shell32)
set_target_properties(KozDbg PROPERTIES SUFFIX "${KOZ_SUFFIX}" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
